# MODEL.1: Python packages for the large audio models the Mac app downloads and runs
# (tandem_desktop/models.py). Installed only for the Mac app, never on the hosted app:
#   python -m pip install -r requirements-models.txt
# Each model that uses the runtime (SEP.1, GEN.2, TIMBRE.1) adds its pinned packages here and
# lists them in its catalog entry's `packages`, so the window can say what is missing.
-r requirements-desktop.txt
# SEP.1: voice separation with Demucs' htdemucs model (MIT). On Linux CI the CPU build of torch is
# installed first from https://download.pytorch.org/whl/cpu; on the Mac the PyPI wheel uses Metal (MPS).
torch==2.15.0
demucs==4.1.0
safetensors==0.8.0
julius==0.2.8
einops==0.8.2
# GEN.2: backing band with MusicGen-Melody (weights CC-BY-NC 4.0, LIC.1) through transformers. Its chroma
# input is computed in tandem_desktop/band.py, since torchaudio has no release for this torch.
transformers==5.22.2
# TIMBRE.1: the voice played as an instrument with RAVE models (CC-BY-NC 4.0, LIC.1) needs only torch and
# julius above: the models are TorchScript, run by torch itself.
# MRT.1: a band from the notes with Google's Magenta RealTime 2 (weights CC-BY 4.0, code Apache 2.0; LIC.1),
# the small model exported for MLX: Apple silicon only. The exported graphs at the pinned revision need MLX
# 0.31.1 exactly (MLX 0.32 refuses them). magenta-rt brings JAX, flax, librosa and a TFLite runtime with it.
magenta-rt[mlx]==2.0.3; sys_platform == "darwin" and platform_machine == "arm64"
mlx==0.31.1; sys_platform == "darwin" and platform_machine == "arm64"
mlx-metal==0.31.1; sys_platform == "darwin" and platform_machine == "arm64"
